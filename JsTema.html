/* ---- Tema: gelap / terang, berputar saat tombol diklik ---- Label tombol menunjukkan tema TUJUAN (berikutnya), bukan tema aktif. Tema kemerdekaan (Merah Putih / 'mp') dinonaktifkan karena sudah lewat Agustus. */ function siapkanHujanBendera_() { const wadah = $('benderaHujan'); if (wadah) wadah.innerHTML = ''; } // siapkanHujanBendera_(); // Dinonaktifkan di luar Agustus const TEMA_URUT = ['dark', 'light']; const TEMA_LABEL = { dark: '🌙 Gelap', light: '☀️ Terang' }; const themeBtn = $('themeBtn'); let temaAktif = 'dark'; function setTheme_(nama) { temaAktif = TEMA_URUT.indexOf(nama) >= 0 ? nama : 'dark'; document.body.classList.remove('mp'); document.body.classList.toggle('light', temaAktif === 'light'); const next = TEMA_URUT[(TEMA_URUT.indexOf(temaAktif) + 1) % TEMA_URUT.length]; themeBtn.textContent = TEMA_LABEL[next]; try { localStorage.setItem('kodomoTema', temaAktif); } catch (e) {} } let temaTersimpan = null; try { temaTersimpan = localStorage.getItem('kodomoTema'); } catch (e) {} setTheme_(temaTersimpan || 'dark'); themeBtn.addEventListener('click', () => setTheme_(TEMA_URUT[(TEMA_URUT.indexOf(temaAktif) + 1) % TEMA_URUT.length]));
