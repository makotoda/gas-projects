function initKodomoLogo3D() { 'use strict'; if (typeof THREE === 'undefined') return; var C = { kayu: 0x5A4632, krem: 0xFCEFD4, emas: 0xFFD35C, emasTua: 0xD69A22, halo: 0xF5B940, setan: 0xF0705A, setanTua: 0xA83A26, tanduk: 0xE4573D, kilauEmas: 0xF5B940, kilauPink: 0xF08AAE }; /* ---------- Tekstur wajah chibi (canvas 256x256) ---------- */ function buatWajah(tipe) { var cv = document.createElement('canvas'); cv.width = cv.height = 256; var tex = new THREE.CanvasTexture(cv); tex.anisotropy = 4; function gambar(kedip) { var x = cv.getContext('2d'); var M = 128; // pusat x.clearRect(0, 0, 256, 256); // Koin + ring pinggir x.beginPath(); x.arc(M, M, 122, 0, 7); x.fillStyle = tipe === 'angel' ? x.moveTo(M - 33, M + 50); x.quadraticCurveTo(M - 17, M + 33, M, M + 50); x.quadraticCurveTo(M + 17, M + 33, M + 33, M + 50); x.lineWidth = 10; x.strokeStyle = '#3A2A18'; x.lineJoin = 'round'; x.stroke(); } tex.needsUpdate = true; } gambar(false); return { tekstur: tex, gambar: gambar }; } /* ---------- Tekstur bintang kilau (sprite) ---------- */ function buatBintang() { var cv = document.createElement('canvas'); cv.width = cv.height = 128; var x = cv.getContext('2d'); x.beginPath(); x.moveTo(64, 4); x.quadraticCurveTo(70, 58, 124, 64); x.quadraticCurveTo(70, 70, 64, 124); x.quadraticCurveTo(58, 70, 4, 64); x.quadraticCurveTo(58, 58, 64, 4); x.fillStyle = '#FFFFFF'; x.fill(); return new THREE.CanvasTexture(cv); } function matte(warna, opsi) { return new THREE.MeshStandardMaterial(Object.assign( { color: warna, roughness: 0.55, metalness: 0.08 }, opsi || {})); } /* ---------- Scene, kamera, cahaya ---------- */ var scene = new THREE.Scene(); var kamera = new THREE.PerspectiveCamera(30, 1, 0.1, 60); kamera.position.set(0, 3.5, 14); kamera.lookAt(0, 2.0, 0); scene.add(new THREE.AmbientLight(0xffffff, 0.55)); scene.add(new THREE.HemisphereLight(0xfff6e8, 0x8880a0, 0.55)); var key = new THREE.DirectionalLight(0xffffff, 0.85); key.position.set(4, 7, 6); scene.add(key); var timbangan = new THREE.Group(); scene.add(timbangan); // Alas + tiang + poros var alas = new THREE.Mesh(new THREE.CylinderGeometry(1.35, 1.6, 0.34, 40), matte(C.kayu)); alas.position.y = 0.17; timbangan.add(alas); porosDot.position.set(0, 3.62, 0.26); timbangan.add(porosDot); // Beam berayun var beam = new THREE.Group(); beam.position.y = 3.62; timbangan.add(beam); var batang = new THREE.Mesh(new THREE.CapsuleGeometry(0.115, 4.7, 6, 20), matte(C.kayu)); batang.rotation.z = Math.PI / 2; beam.add(batang); /* ---------- Satu sisi gantungan: tali + piring + koin ---------- */ function buatGantungan(sisiX, tipe) { var g = new THREE.Group(); // pivot di ujung beam (counter-rotate) g.position.set(sisiX, 0, 0); beam.add(g); var taliMat = matte(C.kayu, { roughness: 0.8 }); var rTepi = 0.92, yPiring = -1.32; // Tali hanya di sisi belakang piring supaya tidak menutupi wajah koin for (var i = 0; i < 3; i++) { var a = Math.PI * (7 / 6 + i / 3); // 210°, 270°, 330° var ujung = new THREE.Vector3(Math.cos(a) * rTepi * 0.92, yPiring + 0.06, Math.sin(a) * rTepi * 0.92); var arah = ujung.clone(); var tali = new THREE.Mesh(new THREE.CylinderGeometry(0.028, 0.028, arah.length(), 8), taliMat); tali.position.copy(ujung).multiplyScalar(0.5); tali.quaternion.setFromUnitVectors(new THREE.Vector3(0, 1, 0), arah.normalize()); g.add(tali); } // Piring: mangkuk setengah bola pipih + bibir var piring = new THREE.Mesh( new THREE.SphereGeometry(rTepi, 40, 16, 0, Math.PI * 2, Math.PI / 2, Math.PI / 2), matte(C.krem, { side: THREE.DoubleSide, roughness: 0.5 })); piring.scale.y = 0.5; piring.position.y = yPiring; g.add(piring); var bibir = new THREE.Mesh(new THREE.TorusGeometry(rTepi, 0.045, 12, 48), matte(C.kayu)); bibir.rotation.x = Math.PI / 2; bibir.position.y = yPiring; g.add(bibir); // Koin chibi menghadap kamera var koin = new THREE.Group(); koin.position.set(0, yPiring + 0.42, 0.05); g.add(koin); var wajah = buatWajah(tipe); var rKoin = 0.72; var sisiMat = matte(tipe === 'angel' ? C.emasTua : C.setanTua, { metalness: 0.3, roughness: 0.35 }); var mukaMat = new THREE.MeshStandardMaterial({ map: wajah.tekstur, metalness: 0.15, roughness: 0.4 }); var badanMat = matte(tipe === 'angel' ? C.emas : C.setan, { metalness: 0.25, roughness: 0.4 }); var silinder = new THREE.Mesh(new THREE.CylinderGeometry(rKoin, rKoin, 0.26, 48), [sisiMat, mukaMat, badanMat]); // Rebahkan tutup silinder ke arah kamera (rot X), lalu putar 90° pada // sumbu Y lokal supaya sumbu-atas tekstur UV tutup (yang default-nya // menunjuk ke samping) jadi tegak vertikal — tanpa ini muka terbaring. silinder.rotation.set(Math.PI / 2, Math.PI / 2, 0); koin.add(silinder); var hasil = { grup: g, koin: koin, wajah: wajah }; if (tipe === 'angel') { var halo = new THREE.Mesh(new THREE.TorusGeometry(0.42, 0.075, 14, 40), matte(C.halo, { metalness: 0.35, roughness: 0.3, emissive: C.halo, emissiveIntensity: 0.25 })); halo.rotation.x = Math.PI / 2.4; halo.position.y = rKoin + 0.3; koin.add(halo); hasil.halo = halo; } else { var tandukGeo = new THREE.ConeGeometry(0.14, 0.4, 16); var tandukMat = matte(C.tanduk, { roughness: 0.45 }); [-1, 1].forEach(function (s) { var t = new THREE.Mesh(tandukGeo, tandukMat); t.position.set(0.5 * s, rKoin - 0.08, 0); t.rotation.z = -0.75 * s; koin.add(t); }); // Ekor: tabung kurva + ujung panah, pivot di sisi kanan koin var ekor = new THREE.Group(); ekor.position.set(rKoin * 0.75, -0.25, -0.12); var kurva = new THREE.QuadraticBezierCurve3( new THREE.Vector3(0, 0, 0), new THREE.Vector3(0.55, -0.15, -0.1), new THREE.Vector3(0.62, -0.75, -0.15)); ekor.add(new THREE.Mesh(new THREE.TubeGeometry(kurva, 20, 0.05, 8), tandukMat)); var ujungEkor = new THREE.Mesh(new THREE.ConeGeometry(0.13, 0.28, 12), tandukMat); ujungEkor.position.set(0.63, -0.86, -0.15); ujungEkor.rotation.z = Math.PI; ekor.add(ujungEkor); koin.add(ekor); hasil.ekor = ekor; } return hasil; } var kiri = buatGantungan(-2.45, 'angel'); var kanan = buatGantungan(2.45, 'devil'); /* ---------- Kilau bintang: emas kiri, pink kanan ---------- */ var texBintang = buatBintang(); var kilau = [ { x: -3.0, y: 4.15, s: 0.6, warna: C.kilauEmas, fase: 0.0 }, { x: -2.9, y: 1.45, s: 0.42, warna: C.kilauEmas, fase: 0.3 }, { x: 3.0, y: 4.3, s: 0.52, warna: C.kilauPink, fase: 0.6 }, { x: 2.95, y: 1.6, s: 0.44, warna: C.kilauPink, fase: 1.1 } ].map(function (k) { var sp = new THREE.Sprite(new THREE.SpriteMaterial({ map: texBintang, color: k.warna, transparent: true })); sp.position.set(k.x, k.y, 0.6); sp.userData = k; timbangan.add(sp); return sp; }); /* ---------- Renderer ---------- */ var renderer = new THREE.WebGLRenderer({ antialias: true, alpha: true }); renderer.setClearColor(0x000000, 0); var kanvas = renderer.domElement; kanvas.style.width = '100%'; kanvas.style.height = '100%'; kanvas.style.display = 'block'; var wadah = null; function pasangKe(container) { wadah = container; container.appendChild(kanvas); ukur(); } function ukur() { if (!wadah) return; var s = Math.max(1, Math.min(wadah.clientWidth, wadah.clientHeight)); renderer.setPixelRatio(Math.min(window.devicePixelRatio || 1, 2)); renderer.setSize(s, s, false); } window.addEventListener('resize', ukur); /* ---------- Animasi ---------- */ var jam = new THREE.Clock(); var kedipSedang = false; var gerakKurang = window.matchMedia && window.matchMedia('(prefers-reduced-motion: reduce)').matches; function frame() { var t = jam.getElapsedTime(); // Sway beam 3.2s (±6°), piring counter-rotate agar menggantung lurus var ayun = Math.sin(t * Math.PI * 2 / 3.2) * 0.105; beam.rotation.z = ayun; kiri.grup.rotation.z = -ayun; kanan.grup.rotation.z = -ayun; // Putaran pelan seluruh timbangan biar kelihatan 3D; koin di-counter // supaya emotnya tetap tegak lurus menghadap depan, tidak ikut miring var yaw = Math.sin(t * 0.4) * 0.32; timbangan.rotation.y = yaw; kiri.koin.rotation.y = -yaw; kanan.koin.rotation.y = -yaw; // Halo mengambang 2.2s kiri.halo.position.y = 1.02 + Math.sin(t * Math.PI * 2 / 2.2) * 0.05; // Ekor goyang 1.4s kanan.ekor.rotation.z = Math.sin(t * Math.PI * 2 / 1.4) * 0.22 + 0.1; // Kilau twinkle 1.8s dengan fase beda-beda kilau.forEach(function (sp) { var k = sp.userData; var w = (Math.sin((t / 1.8 - k.fase) * Math.PI * 2) + 1) / 2; // 0..1 sp.material.opacity = 0.15 + w * 0.85; var s = k.s * (0.55 + w * 0.45); sp.scale.set(s, s, 1); }); // Kedip tiap 4 detik (fase 3.7–3.85) var faseKedip = t % 4; var kedip = faseKedip > 3.7 && faseKedip < 3.85; if (kedip !== kedipSedang) { kedipSedang = kedip; kiri.wajah.gambar(kedip); kanan.wajah.gambar(kedip); } renderer.render(scene, kamera); if (!gerakKurang) requestAnimationFrame(frame); } /* ---------- Penempatan: loader dulu, lalu pindah ke header ---------- */ function keHeader() { var slot = document.getElementById('logoSlot'); if (!slot || wadah === slot) return; slot.textContent = ''; // buang huruf "K" fallback pasangKe(slot); } window.KodomoLogo3D = { keHeader: keHeader, ukur: ukur, pasang: pasangKe }; var loader = document.getElementById('loader'); if (loader && !loader.classList.contains('hide')) { loader.classList.add('logo3d'); pasangKe(document.getElementById('loaderLogo')); } else { keHeader(); // data sudah termuat sebelum Three.js siap } requestAnimationFrame(frame); } /* ---------- Cinematic Landing Page Logic ---------- */ function initLandingPage() { const lp = document.getElementById('landingPage'); if (!lp) return; // Cek preferensi user let skipLp = false; try { skipLp = localStorage.getItem('kodomoSkipLanding') === 'true'; } catch(e) {} if (skipLp) { lp.style.display = 'none'; return; } // --- Canvas Neraka (Efek Api Realistis) --- const cHell = document.getElementById('canvasHell'); const ctxH = cHell.getContext('2d'); let wH, hH; const particlesH = []; function resizeH() { wH = cHell.width = cHell.parentElement.clientWidth; hH = cHell.height = cHell.parentElement.clientHeight; } window.addEventListener('resize', resizeH); resizeH(); class FireParticle { constructor() { this.reset(true); } reset(initial = false) { this.x = Math.random() * wH; this.y = initial ? Math.random() * hH : hH + Math.random() * 50; this.size = Math.random() * 30 + 10; this.speedY = Math.random() * 4 + 2; this.speedX = (Math.random() - 0.5) * 1.5; this.life = initial ? Math.random() : 1; this.decay = Math.random() * 0.015 + 0.005; this.hue = Math.random() * 30; // 0-30 = merah ke oranye } update() { this.y -= this.speedY; this.x += this.speedX; this.life -= this.decay; this.size *= 0.98; if (this.life <= 0 || this.size < 1) this.reset(); } draw() { ctxH.beginPath(); // Buat gradient radial untuk setiap partikel api biar lebih halus const grad = ctxH.createRadialGradient(this.x, this.y, 0, this.x, this.y, this.size); grad.addColorStop(0, `hsla(${this.hue}, 100%, 60%, ${this.life})`); grad.addColorStop(0.4, `hsla(${this.hue}, 100%, 40%, ${this.life * 0.6})`); grad.addColorStop(1, `hsla(${this.hue}, 100%, 20%, 0)`); ctxH.fillStyle = grad; ctxH.arc(this.x, this.y, this.size, 0, Math.PI * 2); ctxH.fill(); } } for(let i=0; i<80; i++) particlesH.push(new FireParticle()); // --- Canvas Surga (Efek Cahaya Ilahi) --- const cHeaven = document.getElementById('canvasHeaven'); const ctxV = cHeaven.getContext('2d'); let wV, hV; const particlesV = []; function resizeV() { wV = cHeaven.width = cHeaven.parentElement.clientWidth; hV = cHeaven.height = cHeaven.parentElement.clientHeight; } window.addEventListener('resize', resizeV); resizeV(); class LightParticle { constructor() { this.reset(true); } reset(initial = false) { this.x = Math.random() * wV; this.y = Math.random() * hV; this.size = Math.random() * 4 + 1; this.speedY = (Math.random() - 0.5) * 0.5 - 0.5; // Mengapung perlahan ke atas this.life = initial ?
